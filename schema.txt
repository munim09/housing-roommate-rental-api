// ============================================================
// USER
// ============================================================

model User {
  id            String     @id @default(uuid()) @db.Uuid

  name          String
  email         String     @unique
  phone         String?    @unique
  password      String

  role          UserRole
  status        UserStatus @default(PENDING_APPROVAL)

  createdAt     DateTime   @default(now())
  updatedAt     DateTime   @updatedAt


  // ----------------------------------------------------------
  // Profiles
  // ----------------------------------------------------------

  ownerProfile   OwnerProfile?
  managerProfile ManagerProfile?
  tenantProfile  TenantProfile?


  // ----------------------------------------------------------
  // Property
  // ----------------------------------------------------------

  createdProperties Property[] @relation("PropertyCreatedBy")


  // ----------------------------------------------------------
  // Ownership
  // ----------------------------------------------------------

  ownedFlats PropertyOwnership[] @relation("FlatOwner")


  // ----------------------------------------------------------
  // Manager assignment
  // ----------------------------------------------------------

  managedFlats ManagerAssignment[] @relation("FlatManager")


  // ----------------------------------------------------------
  // Advertisement
  // ----------------------------------------------------------

  advertisements Advertisement[] @relation("AdvertisementCreatedBy")


  // ----------------------------------------------------------
  // Applications
  // ----------------------------------------------------------

  applications Application[] @relation("ApplicationApplicant")

  reviewedApplications Application[] @relation("ApplicationReviewer")


  // ----------------------------------------------------------
  // Stays
  // ----------------------------------------------------------

  stays Stay[] @relation("StayOccupant")


  // ----------------------------------------------------------
  // Invoices
  // ----------------------------------------------------------

  invoicesAsPayer Invoice[] @relation("InvoicePayer")

  invoicesAsReceiver Invoice[] @relation("InvoiceReceiver")




  // ----------------------------------------------------------
  // Payments
  // ----------------------------------------------------------

  paymentsAsPayer Payment[] @relation("PaymentPayer")

  paymentsAsReceiver Payment[] @relation("PaymentReceiver")


  // ----------------------------------------------------------
  // Viewing requests
  // ----------------------------------------------------------

  viewingRequests ViewingRequest[] @relation("ViewingRequester")

  reviewedViewingRequests ViewingRequest[] @relation("ViewingReviewer")


  // ----------------------------------------------------------
  // Notifications
  // ----------------------------------------------------------

  notifications Notification[]



}

// ============================================================
// FLAT OWNERSHIP
// ============================================================
//
// IMPORTANT:
// Ownership exists ONLY at Flat level.
//
// A Property has no owner.
// A person owns one or more Flats.
//
// Example:
//
// Green View
// ├── Flat A → Owner 1
// ├── Flat B → Owner 2
// └── Flat C → Owner 1
//
// ============================================================

model PropertyOwnership {
  id String @id @default(uuid()) @db.Uuid

  flatId String @db.Uuid
  ownerId String @db.Uuid

  status OwnershipStatus @default(ACTIVE)

  startedAt DateTime @default(now())
  endedAt   DateTime?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt


  flat Flat @relation(
    fields: [flatId],
    references: [id]
  )

  owner User @relation(
    "FlatOwner",
    fields: [ownerId],
    references: [id]
  )


  @@index([flatId])
  @@index([ownerId])
  @@index([status])
}

// ============================================================
// ADVERTISEMENT
// ============================================================
//
// ONE table for:
//
// 1. Owner/Manager rental advertisement
// 2. Tenant roommate advertisement
//
// category:
//   RENTAL
//   ROOMMATE
//
// target:
//   ENTIRE_FLAT
//   ROOM
//
// ============================================================

model Advertisement {
  id String @id @default(uuid()) @db.Uuid

  createdById String @db.Uuid

  flatId String? @db.Uuid
  roomId String? @db.Uuid

  category AdvertisementCategory
  target   AdvertisementTarget

  title       String
  description String?

  monthlyRent Decimal @db.Decimal(12, 2)

  // For roommate advertisement
  availableFrom DateTime?
  availableTo   DateTime?

  status AdvertisementStatus @default(DRAFT)

  publishedAt DateTime?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt


  // ----------------------------------------------------------
  // Relations
  // ----------------------------------------------------------

  createdBy User @relation(
    "AdvertisementCreatedBy",
    fields: [createdById],
    references: [id]
  )

  flat Flat? @relation(
    fields: [flatId],
    references: [id]
  )

  room Room? @relation(
    fields: [roomId],
    references: [id]
  )

  applications Application[]

  viewingRequests ViewingRequest[]


  @@index([createdById])
  @@index([flatId])
  @@index([roomId])
  @@index([category, status])
}

// ============================================================
// STAY
// ============================================================
//
// ONE table for:
//
// Primary tenant stay
// Roommate stay
//
// A roommate is simply another occupant.
//
// ============================================================

model Stay {
  id String @id @default(uuid()) @db.Uuid

  applicationId String @unique @db.Uuid

  occupantId String @db.Uuid

  propertyId String @db.Uuid
  flatId     String @db.Uuid
  roomId     String? @db.Uuid

  type   StayType
  status StayStatus @default(WAITING_FOR_PAYMENT)

  startDate DateTime
  endDate   DateTime

  // Snapshot of rent at approval time.
  // Future advertisement rent changes do not affect this stay.
  monthlyRent Decimal @db.Decimal(12, 2)

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt


  // ----------------------------------------------------------
  // Relations
  // ----------------------------------------------------------

  application Application @relation(
    fields: [applicationId],
    references: [id]
  )

  occupant User @relation(
    "StayOccupant",
    fields: [occupantId],
    references: [id]
  )

  property Property @relation(
    fields: [propertyId],
    references: [id]
  )

  flat Flat @relation(
    fields: [flatId],
    references: [id]
  )

  room Room? @relation(
    fields: [roomId],
    references: [id]
  )

  invoices Invoice[]


  payments Payment[]




  @@index([occupantId])
  @@index([propertyId])
  @@index([flatId])
  @@index([roomId])
  @@index([status])
  @@index([startDate, endDate])
}


// ============================================================
// INVOICE
// ============================================================
//
// ONE table for all rent invoices.
//
// payer / receiver determines the payment direction.
//
// Tenant rent:
//
// Tenant → Owner
//
// Roommate rent:
//
// Roommate → Tenant
//
// ============================================================

model Invoice {
  id String @id @default(uuid()) @db.Uuid

  stayId String @db.Uuid

  payerId    String @db.Uuid
  receiverId String @db.Uuid

  type InvoiceType

  amount Decimal @db.Decimal(12, 2)

  billingPeriodStart DateTime
  billingPeriodEnd   DateTime

  dueDate DateTime

  status BillStatus @default(PENDING)

  description String?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt


  stay Stay @relation(
    fields: [stayId],
    references: [id]
  )

  payer User @relation(
    "InvoicePayer",
    fields: [payerId],
    references: [id]
  )

  receiver User @relation(
    "InvoiceReceiver",
    fields: [receiverId],
    references: [id]
  )

  payments Payment[]


  @@index([stayId])
  @@index([payerId])
  @@index([receiverId])
  @@index([status])
  @@index([dueDate])
}


// ============================================================
// PAYMENT
// ============================================================
//
// ONE payment table.
//
// Examples:
//
// Tenant → Owner
// Tenant → Manager
// Roommate → Tenant
//
// ============================================================

model Payment {
  id String @id @default(uuid()) @db.Uuid

  stayId String? @db.Uuid

  invoiceId     String? @db.Uuid

  payerId    String @db.Uuid
  receiverId String @db.Uuid

  type PaymentType

  amount Decimal @db.Decimal(12, 2)

  status PaymentStatus @default(PENDING)


  transactionReference String?

  gatewayResponse Json?

  paidAt DateTime?

  failureReason String?

  createdAt DateTime @default(now())
  updatedAt DateTime @updatedAt


  // ----------------------------------------------------------
  // Relations
  // ----------------------------------------------------------

  stay Stay? @relation(
    fields: [stayId],
    references: [id]
  )

  invoice Invoice? @relation(
    fields: [invoiceId],
    references: [id]
  )


  payer User @relation(
    "PaymentPayer",
    fields: [payerId],
    references: [id]
  )

  receiver User @relation(
    "PaymentReceiver",
    fields: [receiverId],
    references: [id]
  )


  @@index([stayId])
  @@index([invoiceId])
  @@index([payerId])
  @@index([receiverId])
  @@index([status])
}
